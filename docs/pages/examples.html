Ví dụ TypeScript

Các đoạn mã khởi tạo phổ biến để copy vào component script trong Cocos Creator.

1. TrafficSignals — tự chọn đường rẽ

import { _decorator, Component } from 'cc';
import { SplineJunction, type JunctionEvent } from 'db://cocos-splines/splines/Components/SplineJunction';
const { ccclass, property } = _decorator;
@ccclass('TrafficSignals')
export class TrafficSignals extends Component {
  @property(SplineJunction) junction: SplineJunction | null = null;
  private next = 1;
  onEnable() { this.junction?.AddOnReachEvent(this.onReach, this); }
  onDisable() { this.junction?.RemoveOnReachEvent(this.onReach, this); }
  private onReach(event: JunctionEvent) {
    if (event.entryIndex === 0) {
      event.follower.requestJunctionExit(this.next);
      this.next = this.next === 1 ? 2 : 1;
    } else event.follower.requestJunctionExit(0);
  }
}

2. Chạy làn lệch với tốc độ đều

import { SplineAnimate, AnimateMethod, Easing } from 'db://cocos-splines/splines/Components/SplineAnimate';
const animate = this.getComponent(SplineAnimate)!;
animate.method = AnimateMethod.Speed;
animate.speed = 5;
animate.easing = Easing.Linear;
animate.splineOffset.set(1.8, 0, 0);
animate.constantSpeedWithOffset = true;
animate.restart();
